
# PROJECT STRUCTURE
-----------------------------------------------------------------------------

medical-app/
├── cmd/
│   └── web/
│       ├── main.go          # Wires domain services together & starts server
│       ├── routes.go        # Top-level router & middleware setup
│       └── helpers.go       # Global HTTP response/template helpers
├── internal/
│   ├── auth/                # Auth domain: login, registration, sessions
│   │   ├── handler.go       # Auth HTTP handlers (login/signup pages & posts)
│   │   ├── model.go         # User structs & password hashing
│   │   └── repository.go    # SQL queries for users & sessions
│   ├── appointment/         # Appointment scheduling domain
│   │   ├── handler.go       # Scheduling HTTP handlers
│   │   ├── model.go         # Appointment structs & status enums
│   │   └── repository.go    # SQL queries for booking/fetching appointments
│   ├── record/              # Medical records domain (Doctors only)
│   │   ├── handler.go       # Medical history & report handlers
│   │   ├── model.go         # Record structs
│   │   └── repository.go    # SQL queries for fetching/creating records
│   ├── middleware/          # Cross-cutting HTTP middlewares
│   │   ├── auth.go          # Authentication & RBAC (Doctor vs Patient)
│   │   └── logging.go       # Request logging
│   └── platform/            # Shared infrastructure utilities
│       ├── database/        # DB connection pooling (`sql.DB` setup)
│       └── validator/       # Common form validation logic
├── ui/                      # Frontend assets
│   ├── html/                # Templates grouped by domain or page
│   │   ├── base.tmpl
│   │   ├── auth/            # login.tmpl, register.tmpl
│   │   ├── appointment/     # book.tmpl, list.tmpl
│   │   └── record/          # view_records.tmpl
│   └── static/              # CSS, JavaScript, images
├── migrations/              # Database migration SQL files
├── go.mod
├── go.sum
└── README.md
